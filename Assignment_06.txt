Title: Represent a graph of your college campus using adjacency list /adjacency matrix. Nodes should 
represent the various departments/institutes and links should represent the distance between them. Find 
minimum spanning tree a) Using Kruskal’s algorithm. b) Using Prim’s algorithm. 
 
class Graph: 
    def __init__(self): 
        self.V = 0 
        self.graph = []  # For Kruskal (edge list) 
        self.adj_matrix = [] 
        self.dept_names = [] 
 
    def add_department(self, name): 
        self.dept_names.append(name) 
        self.V = len(self.dept_names) 
        for row in self.adj_matrix: 
            row.append(0) 
        self.adj_matrix.append([0] * self.V) 
 
    def add_edge(self, u, v, w): 
        self.graph.append([u, v, w]) 
        self.adj_matrix[u][v] = w 
        self.adj_matrix[v][u] = w 
  def display_adj_matrix(self): 
        print("\nAdjacency Matrix:") 
        for row in self.adj_matrix: 
            print(row)

    def display_adj_list(self): 
        print("\nAdjacency List:") 
        for i in range(self.V): 
            connections = [] 
            for j in range(self.V): 
                if self.adj_matrix[i][j] != 0: 
                    connections.append(f"{self.dept_names[j]}({self.adj_matrix[i][j]})") 
            print(f"{self.dept_names[i]} -> {', '.join(connections)}") 
    # ---------------- Kruskal's ---------------- 
    def find(self, parent, i): 
        if parent[i] != i: 
            parent[i] = self.find(parent, parent[i]) 
        return parent[i] 
    def union(self, parent, rank, x, y): 
        if rank[x] < rank[y]: 
            parent[x] = y 
        elif rank[x] > rank[y]: 
            parent[y] = x 
        else: 
            parent[y] = x 
            rank[x] += 1

    def prim_mst(self): 
        key = [float('inf')] * self.V 
        parent = [-1] * self.V 
        key[0] = 0 
        mst_set = [False] * self.V 
        for _ in range(self.V): 
            min_key = float('inf') 
            min_index = -1 
            for v in range(self.V): 
                if not mst_set[v] and key[v] < min_key: 
                    min_key = key[v] 
                    min_index = v 
            u = min_index 
            mst_set[u] = True 
            for v in range(self.V): 
                if self.adj_matrix[u][v] and not mst_set[v] and self.adj_matrix[u][v] < key[v]: 
                    key[v] = self.adj_matrix[u][v] 
                    parent[v] = u 
        print("\nMST using Prim's Algorithm:") 
        total_cost = 0 
        for i in range(1, self.V): 
            print(f"{self.dept_names[parent[i]]} - {self.dept_names[i]}: {self.adj_matrix[i][parent[i]]}") 
            total_cost += self.adj_matrix[i][parent[i]] 
        print("Total Distance:", total_cost)
