// Add a student 
void addStudent(Student*& students, int& count, int& capacity) { 
    if (count == capacity) { 
        capacity += 1; 
        students = (Student*) realloc(students, capacity * sizeof(Student)); 
        if (!students) { 
            cout << "Memory allocation failed!\n"; 
            exit(1); 
        } 
    }

void displayStudents(Student* students, int count) { 
    if (count == 0) { 
        cout << "No student records available.\n"; 
        return; 
    } 
 
    cout << "\n--- Student List ---\n"; 
    for (int i = 0; i < count; i++) { 
        cout << "ID: " << students[i].id 
             << ", Name: " << students[i].name 
             << ", CGPA: " << students[i].cgpa << "\n"; 
    } 
} 
 
// Bubble Sort by Name (Alphabetical) 
void bubbleSortByName(Student* students, int count) { 
    for (int i = 0; i < count - 1; i++) { 
        for (int j = 0; j < count - i - 1; j++) { 
            if (strcmp(students[j].name, students[j + 1].name) > 0) { 
                swap(students[j], students[j + 1]); 
            } 
        } 
    } 
} 
 
// Selection Sort by CGPA 
void selectionSortByCGPA(Student* students, int count, bool ascending) { 
    for (int i = 0; i < count - 1; i++) { 
        int minOrMax = i; 
        for (int j = i + 1; j < count; j++) { 
            if ((ascending && students[j].cgpa < students[minOrMax].cgpa) || 
                (!ascending && students[j].cgpa > students[minOrMax].cgpa)) { 
                minOrMax = j; 
            } 
        } 
        if (minOrMax != i) 
            swap(students[i], students[minOrMax]); 
    } 
 
    cout << "Students sorted by CGPA " << (ascending ? "(Ascending)." : "(Descending).") << "\n"; 
}
